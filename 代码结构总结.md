├── snn2/
│   ├── __init__.py                                — 定义 snn2 Python 包入口。
│   ├── artifacts.py                               — 组织 shared Pre/ANN-training bundle、per-run Post-finetuning bundle、selector-aware SNN 路径以及 G/N/T/K 隔离。
│   ├── calibration.py                             — 物化共享 EMA、clamp 后 Phase/MTN（含 global Final RMSNorm）Stage A，并生成 mask-aware、role-specific Stage B Clip profile。
│   ├── config.py                                  — 验证 dual-artifact selector、vanilla selector 限制、ANN/SNN Prefix stage 语义以及 temporal/calibration 配置。
│   ├── controller.py                              — 从独立 Stage A/Clip roots 加载 state，调度 Site 1/7 role Clip 与 clip-free global Final RMSNorm Phase/MTN/GIF topology。
│   ├── conversion.py                              — 按 conversion.use_post_finetuning_artifacts 选择 Prefix/Stage A，校验 source provenance，并生成/验证当前 conversion descriptor。
│   ├── data.py                                    — 构造按 num_samples 隔离的 calibration manifest、固定数据集、tokenization 与 batch collator。
│   ├── evaluation.py                              — 构造 mode-aware static ANN 与 selector-aware temporal SNN controller，并记录 calibration/forward provenance。
│   ├── hadamard.py                                — 封装快速 Hadamard 变换及其规格。
│   ├── logging_utils.py                           — 提供结构化日志与阶段运行记录。
│   ├── model_integration.py                       — 在 repeat 后 merge Site 3/4、PV merge 后执行 Site 6，并安装通用 global Final RMSNorm neuron hook 与严格 saliency 收集。
│   ├── modeling.py                                — 按 stage 解析模型与 Prefix，并将 Final ANN Prefix source 与 Final SNN selector source 解耦。
│   ├── neurons.py                                 — 实现 runtime T/K Phase、MTN 与 Clip，并严格校验 GIF、shared EMA provenance 与 materialized clamp state。
│   ├── phase_statistics.py                        — 定义 Phase/MTN 共享原生 site-layout EMA、group reduction 与 materialized clamp 常量。
│   ├── phase_conversion_regression.py             — 实现支持普通 grouped、attention per-head、Site 5 与 final RMSNorm 的 Phase conversion 回归。
│   ├── prefix.py                                  — 实现 Prefix token 的优化与发现。
│   ├── prefix_cache.py                            — 构造、保存并注入固定 Prefix KV cache。
│   ├── rotation.py                                — 实现 Rotation 规格、权重融合与模型部件访问。
│   ├── sites.py                                   — 定义十个 replacement site、GIF policy、grouping 与 mode-aware Clip eligibility 与连续层拓扑校验。
│   ├── state_validation.py                        — 校验 Stage A/Stage B bundle、global Final RMSNorm Phase/MTN hashes 及 GIF/Clip 禁止条件。
│   ├── stats.py                                   — 按各 site layout 累积 v4 statistics、role-aware saliency 与供 Phase/MTN 共用的 FP32 EMA provenance。
│   ├── temporal_model.py                          — 实现 repeat-KV 后 merged Site 3/4 与 PV merge 后 Site 6 的 full-temporal attention。
│   ├── temporal_ops.py                            — 实现 temporal v8 基础算子并集中定义 GIF provenance、grouping、shared EMA/clamp 与 final-norm topology metadata。
│   └── training.py                                — 实现 ANN replacement-aware 微调并冻结 Stage A 与 selected Stage B profile provenance。
└── tests/

    ├── test_calibration_topology.py               — 验证完整连续层、十 site 与 stale Clip 拒绝。
    ├── test_controller_state_loading.py           — 验证独立 state/clip roots、runtime T/K、role Clip 与 selected-neuron loading。
    ├── test_conversion_metadata.py                — 验证 v14 selector-aware conversion、true/false source selection、aware provenance、schema、工件哈希、Rotation 与无 Clip 合同。

    ├── test_generated_configs.py                  — 动态生成配置并验证 group_size 合法域、temporal v8 与 Site 5 identity policy。
    ├── test_hadamard.py                           — 验证 Hadamard 变换规格与执行行为。
    ├── test_neurons.py                            — 验证 merged/per-head Phase、multi-role/all-low/identity GIF、MTN/Clip 与 strict shape。

    ├── test_temporal_prefix.py                    — 验证位置、Prefix offset 和 time-major 布局。
    ├── test_verify_artifacts.py                    — 验证 Site topology、GIF provenance、Final ANN forward metadata 以及 selector-aware conversion/evaluation source matrix。
