│   ├── materialize_configs.py                     — 从实验矩阵生成并校验十二份运行配置。
│   ├── prepare_data.py                            — 固定训练、验证、评估与 calibration 数据 manifest。
│   ├── prepare_rotation.py                        — 生成并验证 Rotation 工件并保存 fused Base。
│   ├── regress_phase_conversion.py                — 对 static Phase 与 temporal Phase SNN 运行 provenance、单元、trace、ablation 和 locked-token 回归。

│   ├── train_ann.py                               — 启动 ANN 全参数微调并保存最终 checkpoint。
│   └── verify_artifacts.py                        — 验证 Site 2/3/4/6 grouped topology、GIF provenance、conversion provenance 与 ANN/SNN evaluation 完整性。
├── snn2/
│   ├── __init__.py                                — 定义 snn2 Python 包入口。
│   ├── artifacts.py                               — 组织 group_size/num_samples 隔离的 Stage A、Phase/MTN profile、aware ANN 与动态 SNN 工件路径并提供 JSON 和哈希工具。
│   ├── calibration.py                             — 物化 T/K-independent Phase/GIF/MTN Stage A，并生成 mask-aware、role-specific Stage B Clip profile。
│   ├── config.py                                  — 验证 num_samples、固定 phase.base、Phase/MTN runtime 参数与无 max_spikes 的配置协议。
│   ├── controller.py                              — 从独立 Stage A/Clip roots 加载 state，注入 Phase/MTN runtime 参数并调度 Site 1/7 role Clip。
│   ├── conversion.py                              — 校验 Stage A/B 与训练 provenance，并记录 training/deployment T/K 的 conversion descriptor。
│   ├── data.py                                    — 构造按 num_samples 隔离的 calibration manifest、固定数据集、tokenization 与 batch collator。
│   ├── evaluation.py                              — 构造 Stage A/B-aware static ANN 与动态 Phase/MTN SNN controller并记录四类 GIF runtime、site policy、saliency provenance 与 grouping forward metadata。
│   ├── hadamard.py                                — 封装快速 Hadamard 变换及其规格。
│   ├── logging_utils.py                           — 提供结构化日志与阶段运行记录。
│   ├── model_integration.py                       — 在 repeat 后 merge Site 3/4、PV merge 后执行 Site 6，并收集严格 FP32/FP64 role-aware SpikeLLM saliency。
│   ├── modeling.py                                — 加载模型、Rotation、Prefix 与阶段对应状态。
│   ├── neurons.py                                 — 实现 runtime T/K Phase、MTN 与 single/role-specific Clip，以及严格校验 metadata 的 salient、all-low、identity GIF。
│   ├── phase_statistics.py                        — 定义原生 site-layout Phase EMA、channel 与 within-group reduction policy 常量。
│   ├── phase_conversion_regression.py             — 实现支持普通 grouped、attention per-head、Site 5 与 final RMSNorm 的 Phase conversion 回归。
│   ├── prefix.py                                  — 实现 Prefix token 的优化与发现。
│   ├── prefix_cache.py                            — 构造、保存并注入固定 Prefix KV cache。
│   ├── rotation.py                                — 实现 Rotation 规格、权重融合与模型部件访问。
│   ├── sites.py                                   — 定义十个 replacement site、GIF policy、grouping 与 mode-aware Clip eligibility 与连续层拓扑校验。
│   ├── state_validation.py                        — 分别校验 Stage A 无 Clip bundle 与绑定 A 哈希、T、G、num_samples 的 Stage B Clip profile。
│   ├── stats.py                                   — 按 last-dim、attention-head、attention-softmax 布局累积 statistics v3、role-aware FP32/FP64 saliency 与 FP32 Phase EMA。
│   ├── temporal_model.py                          — 实现 repeat-KV 后 merged Site 3/4 与 PV merge 后 Site 6 的 full-temporal attention。
│   ├── temporal_ops.py                            — 实现 temporal v7 基础算子并集中定义 GIF quantization/saliency provenance、grouping、Site 5 identity 与 final-norm policy metadata。
│   └── training.py                                — 实现 ANN replacement-aware 微调并冻结 Stage A 与 selected Stage B profile provenance。
└── tests/

    ├── test_calibration_profiles.py               — 验证 Stage A 的 T/K independence、Stage B profile 隔离及 mask/role-aware Clip。
    ├── test_calibration_topology.py               — 验证完整连续层、十 site 与 stale Clip 拒绝。
    ├── test_controller_state_loading.py           — 验证独立 state/clip roots、runtime T/K、role Clip 与 selected-neuron loading。

    ├── test_generated_configs.py                  — 动态生成配置并验证 group_size 合法域、temporal v7 与 Site 5 identity policy。
    ├── test_hadamard.py                           — 验证 Hadamard 变换规格与执行行为。
    ├── test_neurons.py                            — 验证 merged/per-head Phase、multi-role/all-low/identity GIF、MTN/Clip 与 strict shape。
    ├── test_phase_conversion_regression.py        — 验证 layout-aware Phase 三分解、recorder、bypass、前置条件与 provenance 拒绝。

    ├── test_prefix.py                             — 验证 Prefix token 发现与约束。

    ├── test_sites.py                              — 验证十个 replacement site 定义。

    ├── test_temporal_prefix.py                    — 验证位置、Prefix offset 和 time-major 布局。
    ├── test_verify_artifacts.py                    — 验证 Site 2/3/4 post-repeat、Site 6 merged topology 与 final ANN GIF provenance fail-fast。
    └── test_training.py                           — 验证 ANN 训练 replacement 与 Stage A/B 制品冻结逻辑。
